const { gql } = require("apollo-server");

const typeDefs = gql`
  enum MediaType {
    BOOK
    MOVIE
  }

  interface Media {
    id: ID!
    title: String!
    type: MediaType!
  }

  type Book implements Media {
    id: ID!
    title: String!
    author: String!
    type: MediaType!
    oldField: String @deprecated(reason: "Use 'newField' instead")
  }

  type Movie implements Media {
    id: ID!
    title: String!
    director: String!
    type: MediaType!
  }

  union SearchResult = Book | Movie

  type Query {
    search(term: String!): [SearchResult]
    getBooks: [Book]
    getMovies: [Movie]
  }

  type Mutation {
    addBook(title: String!, author: String!): Book
    addMovie(title: String!, director: String!): Movie
  }
`;

module.exports = typeDefs;
---

const { UserInputError, ApolloError, AuthenticationError } = require("apollo-server");

const books = [
  { id: "1", title: "1984", author: "George Orwell", type: "BOOK" },
];

const movies = [
  { id: "1", title: "Inception", director: "Christopher Nolan", type: "MOVIE" },
];

const resolvers = {
  Query: {
    search: (_, { term }, { user }) => {
      if (!user) throw new AuthenticationError("User not authenticated");
      return [
        ...books.filter((b) => b.title.includes(term)),
        ...movies.filter((m) => m.title.includes(term)),
      ];
    },
    getBooks: () => books,
    getMovies: () => movies,
  },
  Mutation: {
    addBook: (_, { title, author }, { user }) => {
      if (!user) throw new AuthenticationError("User not authenticated");
      if (!title || !author) throw new UserInputError("Missing title/author");
      if (books.find((b) => b.title === title)) {
        throw new ApolloError("Duplicate book", "DUPLICATE_BOOK");
      }
      const newBook = { id: String(books.length + 1), title, author, type: "BOOK" };
      books.push(newBook);
      return newBook;
    },
    addMovie: (_, { title, director }, { user }) => {
      if (!user) throw new AuthenticationError("User not authenticated");
      if (!title || !director) throw new UserInputError("Missing title/director");
      if (movies.find((m) => m.title === title)) {
        throw new ApolloError("Duplicate movie", "DUPLICATE_MOVIE");
      }
      const newMovie = { id: String(movies.length + 1), title, director, type: "MOVIE" };
      movies.push(newMovie);
      return newMovie;
    },
  },
  SearchResult: {
    __resolveType(obj) {
      if (obj.author) return "Book";
      if (obj.director) return "Movie";
      return null;
    },
  },
  Media: {
    __resolveType(obj) {
      if (obj.author) return "Book";
      if (obj.director) return "Movie";
      return null;
    },
  },
};

module.exports = resolvers;
